  <script>
    const sampleData = [
      { EmployeeID: "E001", ชื่อ: "มิน", ตำแหน่ง: "ฝ่ายขาย", ชื่อโครงการ: "เว็บไซต์บริษัท ABC", ยอดราคา: 85000, สถานะ: "กำลังดำเนินการ" },
      { EmployeeID: "E001", ชื่อ: "มิน", ตำแหน่ง: "ฝ่ายขาย", ชื่อโครงการ: "ระบบสมาชิก", ยอดราคา: 42000, สถานะ: "เสนอราคา" },
      { EmployeeID: "E002", ชื่อ: "ต้น", ตำแหน่ง: "นักออกแบบ", ชื่อโครงการ: "เว็บไซต์บริษัท ABC", ยอดราคา: 85000, สถานะ: "กำลังดำเนินการ" },
      { EmployeeID: "E003", ชื่อ: "ฟ้า", ตำแหน่ง: "นักพัฒนา", ชื่อโครงการ: "แอปจัดการสต็อก", ยอดราคา: 120000, สถานะ: "กำลังดำเนินการ" },
      { EmployeeID: "E004", ชื่อ: "นนท์", ตำแหน่ง: "ผู้จัดการโครงการ", ชื่อโครงการ: "ระบบสมาชิก", ยอดราคา: 42000, สถานะ: "เสนอราคา" }
    ];

    const avatars = ["👩🏻‍💼", "👩🏼‍💻", "👩🏼‍💻", "🧑🏻‍💼", "", "🧑🏻‍💼", "🧑🏻‍💼", "👩🏽‍💼"];
    const avatarColors = ["#e7edff", "#ffeedd", "#e4f6ed", "#fce8f0", "#e8f5fb"];
    let rows = sampleData;
    let selectedEmployee = null;

    const peopleEl = document.getElementById("people");
    const searchEl = document.getElementById("searchInput");

    function keyOf(obj, candidates) {
      const entries = Object.keys(obj);
      for (const candidate of candidates) {
        const found = entries.find(k => k.trim().toLowerCase() === candidate.toLowerCase());
        if (found) return obj[found];
      }
      return "";
    }

    function cleanRows(input) {
      return input
        .filter(row => row && Object.values(row).some(v => String(v ?? "").trim() !== ""))
        .map((row, index) => ({
          id: String(keyOf(row, ["EmployeeID", "รหัสพนักงาน", "รหัส"]) || `E${index + 1}`).trim(),
          name: String(keyOf(row, ["ชื่อ", "ชื่อพนักงาน", "EmployeeName", "Name"]) || "ไม่ระบุชื่อ").trim(),
          role: String(keyOf(row, ["ตำแหน่ง", "Position", "Role"]) || "พนักงาน").trim(),
          project: String(keyOf(row, ["ชื่อโครงการ", "โครงการ", "Project", "ProjectName"]) || "ไม่ระบุโครงการ").trim(),
          price: parsePrice(keyOf(row, ["ยอดราคา", "ราคา", "มูลค่า", "Price", "Amount"])),
          status: String(keyOf(row, ["สถานะ", "Status"]) || "ไม่ระบุสถานะ").trim()
        }));
    }

    function parsePrice(value) {
      if (typeof value === "number") return value;
      return Number(String(value ?? "").replace(/[฿,$,\s]/g, "").replace(/,/g, "")) || 0;
    }

    function money(value) {
      return new Intl.NumberFormat("th-TH", {
        style: "currency", currency: "THB", maximumFractionDigits: 0
      }).format(value || 0);
    }

    function groupEmployees() {
      const map = new Map();
      rows.forEach(row => {
        if (!map.has(row.id)) {
          map.set(row.id, { id: row.id, name: row.name, role: row.role, projects: [] });
        }
        map.get(row.id).projects.push(row);
      });
      return [...map.values()];
    }

    function render() {
      const query = searchEl.value.trim().toLowerCase();
      const employees = groupEmployees();
      const filtered = employees.filter(person =>
        `${person.name} ${person.role} ${person.id}`.toLowerCase().includes(query)
      );

      peopleEl.innerHTML = filtered.length ? filtered.map((person, index) => {
        const isSelected = selectedEmployee === person.id;
        const projectLines = person.projects.map(p =>
          `<div class="hover-project">
            <strong>${escapeHtml(p.project)}</strong><br>
            <span class="hover-price">${money(p.price)}</span> · ${escapeHtml(p.status)}
          </div>`
        ).join("");

        return `<article class="person ${isSelected ? "selected" : ""}"
          data-id="${escapeHtml(person.id)}" tabindex="0" role="button"
          aria-label="ดูโครงการของ ${escapeHtml(person.name)}">
          <div class="avatar" style="--avatar-bg:${avatarColors[index % avatarColors.length]}">
            ${avatars[index % avatars.length]}
          </div>
          <div class="person-name">${escapeHtml(person.name)}</div>
          <div class="person-role">${escapeHtml(person.role)}</div>
          <span class="project-count">${person.projects.length} โครงการ</span>
          <div class="hover-card">
            <div class="hover-title">โครงการที่รับผิดชอบ</div>
            ${projectLines || "ยังไม่มีข้อมูลโครงการ"}
          </div>
        </article>`;
      }).join("") : `<div class="empty">ไม่พบพนักงาน ลองค้นหาด้วยชื่ออื่น</div>`;

      document.getElementById("employeeStat").textContent = employees.length;

      // นับและรวมมูลค่าโครงการไม่ซ้ำ โดยใช้ชื่อโครงการเป็นตัวระบุ
      const uniqueProjects = new Map();
      rows.forEach(row => {
        if (!uniqueProjects.has(row.project)) uniqueProjects.set(row.project, row.price);
      });
      document.getElementById("projectStat").textContent = uniqueProjects.size;
      document.getElementById("valueStat").textContent =
        money([...uniqueProjects.values()].reduce((sum, price) => sum + price, 0));

      peopleEl.querySelectorAll(".person").forEach(card => {
        card.addEventListener("click", () => selectEmployee(card.dataset.id));
        card.addEventListener("keydown", event => {
          if (event.key === "Enter" || event.key === " ") {
            event.preventDefault();
            selectEmployee(card.dataset.id);
          }
        });
      });

      if (selectedEmployee && employees.some(person => person.id === selectedEmployee)) {
        showDetails(selectedEmployee);
      }
    }

    function selectEmployee(id) {
      selectedEmployee = id;
      render();
      showDetails(id);
    }

    function showDetails(id) {
      const person = groupEmployees().find(p => p.id === id);
      if (!person) return;

      document.getElementById("detailsTitle").textContent =
        `โครงการของ ${person.name}`;
      document.getElementById("detailsSub").textContent =
        `${person.role} · ${person.projects.length} โครงการ`;

      document.getElementById("detailsTable").innerHTML = `
        <table>
          <thead><tr>
            <th>ชื่อโครงการ</th><th>สถานะ</th><th style="text-align:right">ยอดราคา</th>
          </tr></thead>
          <tbody>
            ${person.projects.map(p => `<tr>
              <td>${escapeHtml(p.project)}</td>
              <td>${escapeHtml(p.status)}</td>
              <td class="price">${money(p.price)}</td>
            </tr>`).join("")}
          </tbody>
        </table>`;
    }

    function escapeHtml(value) {
      return String(value).replace(/[&<>"']/g, char => ({
        "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;", "'": "&#39;"
      })[char]);
    }

    searchEl.addEventListener("input", render);

    document.getElementById("fileInput").addEventListener("change", async event => {
      const file = event.target.files[0];
      if (!file) return;

      try {
        const data = await file.arrayBuffer();
        const workbook = XLSX.read(data, { type: "array" });
        const sheet = workbook.Sheets[workbook.SheetNames[0]];
        const parsed = XLSX.utils.sheet_to_json(sheet, { defval: "" });
        rows = cleanRows(parsed);
        selectedEmployee = null;
        document.getElementById("message").textContent =
          rows.length ? `โหลดข้อมูลจาก ${file.name} แล้ว` : "ไม่พบข้อมูลในชีตแรกของไฟล์";
        document.getElementById("detailsTitle").textContent = "รายละเอียดโครงการ";
        document.getElementById("detailsSub").textContent = "เลือกพนักงานเพื่อดูรายการโครงการ";
        document.getElementById("detailsTable").innerHTML = "";
        render();
      } catch (error) {
        document.getElementById("message").textContent =
          "อ่านไฟล์ไม่สำเร็จ โปรดตรวจสอบว่าเป็นไฟล์ Excel หรือ CSV ที่ถูกต้อง";
      }
    });

    render();
  </script>
